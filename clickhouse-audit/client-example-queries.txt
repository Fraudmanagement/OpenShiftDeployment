session_log configuration
-------------------------
/etc/clickhouse-server/config.d/session-log.xml

<clickhouse>
    <session_log>
        <database>system</database>
        <table>session_log</table>
        <flush_interval_milliseconds>7500</flush_interval_milliseconds>
    </session_log>
</clickhouse>

sudo systemctl restart clickhouse-server

if an existing setup, add the following and proceed
---------------------------------------------------
sudo tee /etc/clickhouse-server/config.d/fraudbuster-session-log.xml > /dev/null <<'EOF'
<clickhouse>
    <session_log>
        <database>system</database>
        <table>session_log</table>
        <flush_interval_milliseconds>7500</flush_interval_milliseconds>
    </session_log>
</clickhouse>
EOF

sudo chown clickhouse:clickhouse /etc/clickhouse-server/config.d/fraudbuster-session-log.xml
sudo systemctl restart clickhouse-server
